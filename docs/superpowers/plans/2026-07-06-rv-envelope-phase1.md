# RV Envelope (Phase 1) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add an `rv-class-a` scene template that seeds a 40′ Class A motorhome envelope — two levels (chassis + main), a roof reference level, a locked exterior boundary, and the named basement bays — so all later RV systems have a coordinate space to live in.

**Architecture:** Pure additive template, mirroring `packages/mcp/src/templates/empty-studio.ts`. No new node types: the envelope is built from existing `site` / `building` / `level` / `wall` / `zone` nodes. Bays are `zone` nodes tagged through their `metadata` (`bayRole`); the fixed envelope is marked with `metadata.locked`. The template is a deterministic `SceneGraph` whose ids are regenerated by `cloneSceneGraph` when instantiated.

**Tech Stack:** TypeScript, Zod schemas (`@pascal-app/core/schema`), `bun:test`, the MCP template registry (`packages/mcp/src/templates`).

## Global Constraints

- Toolchain is **bun only** — build with `bun`, test with `bun test`. (`PATH` must include `~/.bun/bin`.)
- Every template node MUST pass `AnyNode.safeParse` (the generic `templates.test.ts` enforces this once registered).
- Deterministic placeholder ids (e.g. `site_rv`, `level_main`) — regenerated by `cloneSceneGraph` at instantiation. Do not use random ids.
- Nominal coach dimensions live in named constants so real measurements can replace them later: length 40′ = **12.192 m**, exterior width 102″ = **2.5908 m**, interior wall height **2.0 m**.
- No new node types in this phase. If a field seems missing, store it in the node's `metadata`, not a schema change.
- Follow the existing template file shape exactly: build nodes → assemble `nodes` record + `rootNodeIds` → export `template: SceneGraph` and `metadata`.

Create `packages/mcp/src/templates/rv-class-a.ts`:

```typescript
import type { SceneGraph } from '@pascal-app/core/clone-scene-graph'
import type { AnyNode, AnyNodeId } from '@pascal-app/core/schema'

/**
 * 40′ Class A motorhome envelope. Deterministic ids (`site_rv`, `level_main`,
 * …) are regenerated by `cloneSceneGraph` before the template is applied.
 *
 * Coordinate frame: XZ plane, origin at coach center. Width (X) is the exterior
 * body width; length (Z) runs front (−Z) to rear (+Z). Three levels stack in the
 * building: chassis (pass-through bays, level −1), main (living floor, level 0),
 * roof (reference plane for A/C, solar, vents, level 1).
 */

const HALF_W = 2.5908 / 2 // exterior body half-width (102")
const HALF_L = 12.192 / 2 // coach half-length (40')
const WALL_HEIGHT = 2.0 // interior wall height
const WALL_THICKNESS = 0.05

// Bay boundaries along Z, front (−) to rear (+): 7 pass-through compartments.
const BAY_BANDS: Array<{ role: string; name: string; z0: number; z1: number }> = [
  { role: 'generator', name: 'Generator bay', z0: -HALF_L, z1: -4.5 },
  { role: 'battery', name: 'Battery bay', z0: -4.5, z1: -3.3 },
  { role: 'propane', name: 'Propane bay', z0: -3.3, z1: -2.1 },
  { role: 'fresh-tank', name: 'Fresh water tank', z0: -2.1, z1: -0.5 },
  { role: 'grey-tank', name: 'Grey tank', z0: -0.5, z1: 1.0 },
  { role: 'black-tank', name: 'Black tank', z0: 1.0, z1: 2.5 },
  { role: 'engine', name: 'Engine bay', z0: 2.5, z1: HALF_L },
]

function buildNodes(): AnyNode[] {
  const site = {
    object: 'node',
    id: 'site_rv' as AnyNodeId,
    type: 'site',
    parentId: null,
    visible: true,
    metadata: {},
    polygon: {
      type: 'polygon',
      points: [
        [-8, -8],
        [8, -8],
        [8, 8],
        [-8, 8],
      ],
    },
    children: ['building_rv' as AnyNodeId],
  } as unknown as AnyNode

  const building = {
    object: 'node',
    id: 'building_rv' as AnyNodeId,
    type: 'building',
    parentId: 'site_rv' as AnyNodeId,
    visible: true,
    metadata: {},
    position: [0, 0, 0],
    rotation: [0, 0, 0],
    children: ['level_chassis', 'level_main', 'level_roof'] as AnyNodeId[],
  } as unknown as AnyNode

  const wallDefs = [
    { id: 'wall_n', start: [-HALF_W, -HALF_L], end: [HALF_W, -HALF_L] },
    { id: 'wall_e', start: [HALF_W, -HALF_L], end: [HALF_W, HALF_L] },
    { id: 'wall_s', start: [HALF_W, HALF_L], end: [-HALF_W, HALF_L] },
    { id: 'wall_w', start: [-HALF_W, HALF_L], end: [-HALF_W, -HALF_L] },
  ]
  const walls = wallDefs.map(
    (w) =>
      ({
        object: 'node',
        id: w.id as AnyNodeId,
        type: 'wall',
        parentId: 'level_main' as AnyNodeId,
        visible: true,
        metadata: { locked: true },
        children: [],
        thickness: WALL_THICKNESS,
        height: WALL_HEIGHT,
        start: w.start,
        end: w.end,
        frontSide: 'unknown',
        backSide: 'unknown',
      }) as unknown as AnyNode,
  )

  const bays = BAY_BANDS.map(
    (b) =>
      ({
        object: 'node',
        id: `bay_${b.role}` as AnyNodeId,
        type: 'zone',
        parentId: 'level_chassis' as AnyNodeId,
        visible: true,
        metadata: { bayRole: b.role, locked: true },
        name: b.name,
        color: '#94a3b8',
        polygon: [
          [-HALF_W, b.z0],
          [HALF_W, b.z0],
          [HALF_W, b.z1],
          [-HALF_W, b.z1],
        ],
      }) as unknown as AnyNode,
  )

  const levelChassis = {
    object: 'node',
    id: 'level_chassis' as AnyNodeId,
    type: 'level',
    parentId: 'building_rv' as AnyNodeId,
    visible: true,
    metadata: {},
    level: -1,
    children: bays.map((b) => b.id) as AnyNodeId[],
  } as unknown as AnyNode

  const levelMain = {
    object: 'node',
    id: 'level_main' as AnyNodeId,
    type: 'level',
    parentId: 'building_rv' as AnyNodeId,
    visible: true,
    metadata: {},
    level: 0,
    children: walls.map((w) => w.id) as AnyNodeId[],
  } as unknown as AnyNode

  const levelRoof = {
    object: 'node',
    id: 'level_roof' as AnyNodeId,
    type: 'level',
    parentId: 'building_rv' as AnyNodeId,
    visible: true,
    metadata: {},
    level: 1,
    children: [],
  } as unknown as AnyNode

  return [site, building, levelChassis, levelMain, levelRoof, ...walls, ...bays]
}

function buildTemplate(): SceneGraph {
  const nodes: Record<AnyNodeId, AnyNode> = {}
  for (const node of buildNodes()) {
    nodes[node.id as AnyNodeId] = node
  }
  return { nodes, rootNodeIds: ['site_rv'] as AnyNodeId[] }
}

export const template: SceneGraph = buildTemplate()

export const metadata = {
  id: 'rv-class-a',
  name: '40′ Class A motorhome',
  description:
    'RV envelope: chassis (pass-through bays), main living level, and roof reference plane, with a locked 40′ × 102″ exterior boundary and seven named basement bays.',
} as const
```
